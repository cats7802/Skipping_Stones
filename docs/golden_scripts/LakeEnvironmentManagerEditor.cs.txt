#if UNITY_EDITOR
using UnityEditor;
using UnityEngine;

[CustomEditor(typeof(LakeEnvironmentManager))]
public class LakeEnvironmentManagerEditor : Editor
{
    private SerializedProperty mapTitleProp;
    private SerializedProperty mapThumbnailProp;
    private SerializedProperty startMapPrefabProp;
    private SerializedProperty loopSlotsProp;
    private SerializedProperty loopRepeatCountProp;
    private SerializedProperty endingTriggerModeProp;
    private SerializedProperty endingMapPrefabProp;
    private SerializedProperty targetClearDistanceProp;
    private SerializedProperty stopSpawningOnEndingProp;
    private SerializedProperty autoChunkSizeProp;

    private void OnEnable()
    {
        mapTitleProp = serializedObject.FindProperty("mapTitle");
        mapThumbnailProp = serializedObject.FindProperty("mapThumbnail");
        startMapPrefabProp = serializedObject.FindProperty("startMapPrefab");
        loopSlotsProp = serializedObject.FindProperty("loopSlots");
        loopRepeatCountProp = serializedObject.FindProperty("loopRepeatCount");
        endingTriggerModeProp = serializedObject.FindProperty("endingTriggerMode");
        endingMapPrefabProp = serializedObject.FindProperty("endingMapPrefab");
        targetClearDistanceProp = serializedObject.FindProperty("targetClearDistance");
        stopSpawningOnEndingProp = serializedObject.FindProperty("stopSpawningOnEnding");
        autoChunkSizeProp = serializedObject.FindProperty("autoChunkSize");
    }

    public override void OnInspectorGUI()
    {
        serializedObject.Update();
        LakeEnvironmentManager mgr = (LakeEnvironmentManager)target;

        GUIStyle headerStyle = new GUIStyle(EditorStyles.boldLabel)
        {
            fontSize = 12,
            alignment = TextAnchor.MiddleLeft
        };

        EditorGUILayout.Space(4);
        EditorGUILayout.LabelField("🏷️ 0. 맵 메타 정보", headerStyle);
        EditorGUILayout.PropertyField(mapTitleProp, new GUIContent("Map Title (맵 이름)", "3번 맵 선택 및 로비/인게임 UI에 표시될 맵 타이틀 텍스트"));
        EditorGUILayout.PropertyField(mapThumbnailProp, new GUIContent("Map Thumbnail (2D)", "3번 맵 선택 및 로비 UI에 표시될 2D 맵 썸네일 이미지"));
        EditorGUILayout.Space(6);

        EditorGUILayout.LabelField("🎬 1. 모듈러 스토리 시퀀스 설정", headerStyle);

        // 1. 시작 맵
        EditorGUILayout.PropertyField(startMapPrefabProp, new GUIContent("Start Map Prefab (SM)", "시작 전용 맵 프리팹 (비어있을 시 슬롯 1번 BaseMap 사용)"));
        EditorGUILayout.Space(6);

        // 2. 루프 슬롯 스핀박스 및 직관적 슬롯 에디터
        EditorGUILayout.LabelField("🔁 2. 중간 루프 슬롯 구성 (Loop Backbone)", headerStyle);

        int currentCount = loopSlotsProp.arraySize;
        EditorGUILayout.BeginHorizontal();
        EditorGUILayout.PrefixLabel(new GUIContent("루프 맵 개수 (슬롯 수)", "순환할 루프 슬롯의 총 개수"));
        int newCount = EditorGUILayout.IntField(currentCount, GUILayout.Width(50));
        
        if (GUILayout.Button("▲", GUILayout.Width(24))) newCount++;
        if (GUILayout.Button("▼", GUILayout.Width(24))) newCount--;
        EditorGUILayout.EndHorizontal();

        if (newCount < 0) newCount = 0;
        if (newCount != currentCount)
        {
            loopSlotsProp.arraySize = newCount;
        }

        EditorGUI.indentLevel++;
        for (int i = 0; i < loopSlotsProp.arraySize; i++)
        {
            SerializedProperty slotProp = loopSlotsProp.GetArrayElementAtIndex(i);
            SerializedProperty baseMapProp = slotProp.FindPropertyRelative("baseMapPrefab");
            SerializedProperty useVarProp = slotProp.FindPropertyRelative("useVariations");
            SerializedProperty varListProp = slotProp.FindPropertyRelative("variationPrefabs");

            EditorGUILayout.BeginVertical(EditorStyles.helpBox);
            EditorGUILayout.LabelField($"📌 [ 슬롯 {i + 1} ]", EditorStyles.boldLabel);

            EditorGUILayout.PropertyField(baseMapProp, new GUIContent("Base Map Prefab", "이 슬롯의 기준 메인 프리팹 (필수)"));

            EditorGUILayout.PropertyField(useVarProp, new GUIContent("변주(랜덤) 맵 사용", "체크 시 등록된 변주 프리팹들 중 무작위 선택 (슬롯 순서는 유지)"));

            if (useVarProp.boolValue)
            {
                EditorGUI.indentLevel++;
                EditorGUILayout.PropertyField(varListProp, new GUIContent("추가 변주 프리팹 목록"), true);
                EditorGUI.indentLevel--;
            }

            EditorGUILayout.EndVertical();
            EditorGUILayout.Space(2);
        }
        EditorGUI.indentLevel--;

        EditorGUILayout.Space(6);
        // 3. 엔딩 및 루프 반복 규칙
        EditorGUILayout.LabelField("🏁 3. 루프 횟수 & 피날레 엔딩 규칙", headerStyle);
        EditorGUILayout.BeginVertical(EditorStyles.helpBox);
        EditorGUILayout.PropertyField(endingTriggerModeProp, new GUIContent("엔딩 판정 방식 (Trigger Mode)"));

        if ((LakeEnvironmentManager.EndingTriggerMode)endingTriggerModeProp.enumValueIndex == LakeEnvironmentManager.EndingTriggerMode.ByLoopCount)
        {
            EditorGUILayout.PropertyField(loopRepeatCountProp, new GUIContent("루프 반복 횟수 (Repeat Count)", "중간 슬롯(1~N번) 시퀀스를 몇 회 반복할 것인가? (1 = 1회 완주 후 엔딩 맵 스폰, 0 = 무한 루프)"));
        }
        else if ((LakeEnvironmentManager.EndingTriggerMode)endingTriggerModeProp.enumValueIndex == LakeEnvironmentManager.EndingTriggerMode.ByDistance)
        {
            EditorGUILayout.PropertyField(targetClearDistanceProp, new GUIContent("목표 거리 (Target Clear)", "0 이하일 경우 엔딩 없이 무한 루프"));
        }

        EditorGUILayout.PropertyField(endingMapPrefabProp, new GUIContent("Ending Map Prefab (EM)", "목표 거리/루프 횟수 도달 시 등장할 엔딩 맵 프리팹"));
        EditorGUILayout.PropertyField(stopSpawningOnEndingProp, new GUIContent("엔딩 도달 시 스폰 정지 (완주)", "체크 시 엔딩 맵 뒤로 추가 청크 생성을 멈추고 완주 상태로 종료"));
        EditorGUILayout.EndVertical();

        EditorGUILayout.Space(6);
        // 4. 청크 측정 정보
        EditorGUILayout.LabelField("📏 4. 스트리밍 정보", headerStyle);
        EditorGUILayout.PropertyField(autoChunkSizeProp, new GUIContent("Auto Chunk Size (Z)", "배경 1개 청크의 Z축 기본 기준 길이"));

        serializedObject.ApplyModifiedProperties();

        // 5. 에디터 맵 생성 테스트 도구
        EditorGUILayout.Space(10);
        EditorGUILayout.LabelField("🧪 5. 맵 시퀀스 테스트 도구 (In-Editor Test)", headerStyle);
        EditorGUILayout.BeginVertical(EditorStyles.helpBox);

        GUI.backgroundColor = new Color(0.35f, 0.85f, 0.45f);
        if (GUILayout.Button("▶️ 규칙대로 전체 시퀀스 맵 일괄 생성 (Build Full Sequence)", GUILayout.Height(34)))
        {
            Undo.RegisterFullObjectHierarchyUndo(mgr.gameObject, "Build Full Sequence Map");
            mgr.TestBuildFullSequence();
            SceneView.RepaintAll();
        }
        GUI.backgroundColor = Color.white;

        EditorGUILayout.Space(3);
        EditorGUILayout.BeginHorizontal();
        if (GUILayout.Button("➕ 다음 1개 청크 스폰 (Spawn Next)", GUILayout.Height(26)))
        {
            Undo.RegisterFullObjectHierarchyUndo(mgr.gameObject, "Spawn Next Chunk");
            mgr.TestSpawnNextChunk();
            SceneView.RepaintAll();
        }

        GUI.backgroundColor = new Color(1f, 0.5f, 0.5f);
        if (GUILayout.Button("🧹 생성된 청크 초기화 (Clear All)", GUILayout.Height(26)))
        {
            Undo.RegisterFullObjectHierarchyUndo(mgr.gameObject, "Clear Test Chunks");
            mgr.TestClearChunks();
            SceneView.RepaintAll();
        }
        GUI.backgroundColor = Color.white;
        EditorGUILayout.EndHorizontal();

        EditorGUILayout.Space(4);
        GUI.backgroundColor = new Color(0.25f, 0.80f, 1.0f);
        if (GUILayout.Button("🌊 등록된 모든 맵 강줄기 곡선 일괄 베이킹 (Bake Splines)", GUILayout.Height(30)))
        {
            SkippingStones.EditorTools.RiverPathBakerWindow.BakeAllFromEnvManager(mgr.gameObject);
        }
        GUI.backgroundColor = Color.white;

        EditorGUILayout.HelpBox(
            "💡 [안내]\n" +
            "• '전체 시퀀스 맵 생성': Start ➔ LoopSlots (N회) ➔ Ending 맵이 소켓 앵커로 도킹되어 에디터 씬에 즉시 펼쳐집니다.\n" +
            "• '강줄기 곡선 일괄 베이킹': 등록된 모든 맵의 물길 곡선 & 강폭을 계산하여 프리팹 원본에 영구 저장합니다.", MessageType.Info);
        EditorGUILayout.EndVertical();
    }
}
#endif
