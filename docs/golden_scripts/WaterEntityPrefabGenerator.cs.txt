using UnityEngine;
using UnityEditor;
using System.IO;

public static class WaterEntityPrefabGenerator
{
    [MenuItem("Tools/Generate Water Entity Prefabs")]
    public static void GenerateAllPrefabs()
    {
        string dir = "Assets/Resources";
        if (!Directory.Exists(dir))
        {
            Directory.CreateDirectory(dir);
        }

        CreateBoostPadPrefab(dir);
        CreateObstacleRockPrefab(dir);
        CreateTargetZonePrefab(dir);
        CreateFriendFlagPrefab(dir);
        CreateJumpingFishPrefab(dir);
        CreateLilyClusterPrefab(dir);

        AssetDatabase.SaveAssets();
        AssetDatabase.Refresh();
        Debug.Log("✅ [WaterEntityPrefabGenerator] 모든 수면 엔티티 프리팹 6종이 Assets/Resources/ 에 정상 생성되었습니다!");
    }

    private static Material GetLitMaterial(Color baseColor, Color? emission = null)
    {
        Shader shader = Shader.Find("Universal Render Pipeline/Lit") ?? Shader.Find("Standard");
        Material mat = new Material(shader);
        mat.SetColor("_BaseColor", baseColor);
        if (emission.HasValue && mat.HasProperty("_EmissionColor"))
        {
            mat.EnableKeyword("_EMISSION");
            mat.SetColor("_EmissionColor", emission.Value);
        }
        return mat;
    }

    private static void CreateBoostPadPrefab(string dir)
    {
        GameObject root = new GameObject("BoostPad");
        BoostPad bp = root.AddComponent<BoostPad>();

        GameObject pad = GameObject.CreatePrimitive(PrimitiveType.Cube);
        pad.name = "PadMesh";
        pad.transform.SetParent(root.transform);
        pad.transform.localPosition = new Vector3(0f, 0.05f, 0f);
        pad.transform.localScale = new Vector3(3.5f, 0.08f, 5.0f);
        pad.GetComponent<Renderer>().sharedMaterial = GetLitMaterial(new Color(0.1f, 1f, 0.6f, 0.9f), new Color(0.1f, 1f, 0.6f) * 1.5f);
        Object.DestroyImmediate(pad.GetComponent<Collider>());

        BoxCollider col = root.AddComponent<BoxCollider>();
        col.isTrigger = true;
        col.size = new Vector3(4f, 1.5f, 5.5f);
        col.center = new Vector3(0f, 0.5f, 0f);

        string path = $"{dir}/BoostPad.prefab";
        PrefabUtility.SaveAsPrefabAsset(root, path);
        Object.DestroyImmediate(root);
    }

    private static void CreateObstacleRockPrefab(string dir)
    {
        GameObject root = new GameObject("ObstacleRock");
        root.AddComponent<ObstacleRock>();

        GameObject rock = GameObject.CreatePrimitive(PrimitiveType.Sphere);
        rock.name = "RockModel";
        rock.transform.SetParent(root.transform);
        rock.transform.localPosition = new Vector3(0f, 0.4f, 0f);
        rock.transform.localScale = new Vector3(2.4f, 1.6f, 2.4f);
        rock.GetComponent<Renderer>().sharedMaterial = GetLitMaterial(new Color(0.35f, 0.32f, 0.30f));
        
        SphereCollider sc = rock.GetComponent<SphereCollider>();
        if (sc != null) sc.isTrigger = true;

        string path = $"{dir}/ObstacleRock.prefab";
        PrefabUtility.SaveAsPrefabAsset(root, path);
        Object.DestroyImmediate(root);
    }

    private static void CreateTargetZonePrefab(string dir)
    {
        GameObject root = new GameObject("TargetZone");
        root.AddComponent<FloatingTargetZone>();

        // 1. 외곽 링
        GameObject outer = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
        outer.name = "OuterRing";
        outer.transform.SetParent(root.transform);
        outer.transform.localPosition = new Vector3(0f, 0.02f, 0f);
        outer.transform.localScale = new Vector3(7f, 0.02f, 7f);
        outer.GetComponent<Renderer>().sharedMaterial = GetLitMaterial(new Color(1f, 0.85f, 0.1f, 0.9f), new Color(1f, 0.85f, 0.1f) * 0.5f);
        Object.DestroyImmediate(outer.GetComponent<Collider>());

        // 2. 중간 링
        GameObject mid = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
        mid.name = "MidRing";
        mid.transform.SetParent(root.transform);
        mid.transform.localPosition = new Vector3(0f, 0.03f, 0f);
        mid.transform.localScale = new Vector3(4.5f, 0.02f, 4.5f);
        mid.GetComponent<Renderer>().sharedMaterial = GetLitMaterial(new Color(1f, 0.25f, 0.25f, 0.9f), new Color(1f, 0.25f, 0.25f) * 0.8f);
        Object.DestroyImmediate(mid.GetComponent<Collider>());

        // 3. 중심
        GameObject center = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
        center.name = "BullseyeCenter";
        center.transform.SetParent(root.transform);
        center.transform.localPosition = new Vector3(0f, 0.04f, 0f);
        center.transform.localScale = new Vector3(2.1f, 0.02f, 2.1f);
        center.GetComponent<Renderer>().sharedMaterial = GetLitMaterial(new Color(0.1f, 0.9f, 1f, 0.95f), new Color(0.1f, 0.9f, 1f) * 1.5f);
        Object.DestroyImmediate(center.GetComponent<Collider>());

        SphereCollider col = root.AddComponent<SphereCollider>();
        col.isTrigger = true;
        col.radius = 3.5f;

        string path = $"{dir}/TargetZone.prefab";
        PrefabUtility.SaveAsPrefabAsset(root, path);
        Object.DestroyImmediate(root);
    }

    private static void CreateJumpingFishPrefab(string dir)

    }

    private static void CreateLilyClusterPrefab(string dir)
    {
        GameObject root = new GameObject("LilyPadCluster");

        Material lilyMat = GetLitMaterial(new Color(0.12f, 0.58f, 0.28f));
        Material flowerMat = GetLitMaterial(new Color(0.98f, 0.65f, 0.85f), new Color(0.98f, 0.65f, 0.85f) * 0.5f);

        Vector3[] offsets = new Vector3[] {
            new Vector3(-0.8f, 0f, -0.6f),
            new Vector3(0.7f, 0f, -0.5f),
            new Vector3(-0.4f, 0f, 0.8f),
            new Vector3(0.8f, 0f, 0.6f),
            new Vector3(0f, 0f, 0f)
        };

        for (int i = 0; i < offsets.Length; i++)
        {
            GameObject pad = GameObject.CreatePrimitive(PrimitiveType.Cylinder);
            pad.name = $"Pad_{i}";
            pad.transform.SetParent(root.transform);
            pad.transform.localPosition = offsets[i];
            pad.transform.localScale = new Vector3(1.6f, 0.02f, 1.6f);
            pad.GetComponent<Renderer>().sharedMaterial = lilyMat;
            Object.DestroyImmediate(pad.GetComponent<Collider>());

            if (i == offsets.Length - 1)
            {
                GameObject flower = GameObject.CreatePrimitive(PrimitiveType.Sphere);
                flower.name = "LotusFlower";
                flower.transform.SetParent(pad.transform);
                flower.transform.localPosition = new Vector3(0f, 0.6f, 0f);
                flower.transform.localScale = new Vector3(0.4f, 0.8f, 0.4f);
                flower.GetComponent<Renderer>().sharedMaterial = flowerMat;
                Object.DestroyImmediate(flower.GetComponent<Collider>());
            }
        }

        string path = $"{dir}/LilyPadCluster.prefab";
        PrefabUtility.SaveAsPrefabAsset(root, path);
        Object.DestroyImmediate(root);
    }
}
